from pathlib import Path
import json
import pandas as pd

root = Path(
    "/workspace/exp_v2/drivemem_memory/experiments/pair_utility_v1"
)
fit = json.loads((root / "fit.json").read_text())

    k: fit[k] for k in ("primary", "h", "selected", "cv")
}, indent=2))

for cohort in ("budget20gb_v1", "validation_cached4_v1"):
    df = pd.read_csv(
        root / f"{cohort}_per_scene.csv", dtype={"token": str}
    )
    gains = df.pivot_table(
        index="session_id", columns="method",
        values="raw_gain_pp", aggfunc="mean",
    )
    gains["memory_minus_no_memory"] = (
        gains["direct_memory"] - gains["direct_no_memory"]
    )
    print(f"\nGAIN BY SESSION: {cohort}")
    print(gains.to_string())

    print("\nLEAVE-ONE-SESSION-OUT:")
    for sid in gains.index:
        rest = gains.drop(index=sid)
        print(
            sid,
            "memory_vs_baseline=", round(rest.direct_memory.mean(), 6),
            "memory_vs_control=",
            round(rest.memory_minus_no_memory.mean(), 6),
        )

    q = df[df.method == "direct_memory"]
    columns = [
        "token", "session_id", "candidate_id",
        "raw_gain_pp", "raw_harm", "raw_quality_degraded",
    ] + [c for c in df if c.startswith("raw_delta_")]
    print("\nNON-NEUTRAL MEMORY DECISIONS:")
    print(q.loc[
        (q.raw_gain_pp.abs() > 1e-6) | q.raw_harm,
        columns,
    ].to_string(index=False))
